# =====================================================================================================================
# KCD2_TPVCamera_rules.txt - how TPVCamera's third-person camera treats each object in the world
#
# These rules are built into TPVCamera, which needs no file for them and does not ship this one. Every file named here
# sits beside KCD2_TPVCamera.asi, and "this file" means the rules in force: the built-in ones, or your own copy.
# What the rules say is final: nothing in the mod adds to them or overrides them. An object that no rule matches gets
# the @default rule (in the "Everything else" group below). Without a @default line, TPVCamera leaves such an object
# alone, as if its rule said exclude, and KCD2_TPVCamera.log warns about it. The rules cannot change which collision
# the camera meets in the first place (see WHAT THE RULES CANNOT CHANGE).
#
# TO READ THEM: [Advanced] ExportRules = true in KCD2_TPVCamera.ini writes them to KCD2_TPVCamera_rules.default.txt,
#   line for line, a copy TPVCamera never reads. In the log, 'built-in KCD2_TPVCamera_rules.txt:295' is its line 295.
# TO CHANGE THEM: write your lines in KCD2_TPVCamera_rules_User.txt (see YOUR OWN FILE AND PATCHES). Updates never
#   touch it, and at the same level its lines win over this file's, whatever the length of their match.
# TO REPLACE THEM: rename that copy to KCD2_TPVCamera_rules.txt. TPVCamera then reads it instead, and it is final: the
#   built-in rules of later updates no longer reach you. Delete it to go back to them, also while the game runs. If it
#   cannot be read, TPVCamera uses the built-in rules and says so in KCD2_TPVCamera.log.
# ---------------------------------------------------------------------------------------------------------------------
# ONE RULE PER LINE
#
#     <match> = <behaviour> [options]
#
#   Text after # is a comment. Upper and lower case are the same. Write model paths with / as the game does. A match
#   cannot hold a space. Save the file as UTF-8, ANSI or UTF-16 text (Notepad's "Unicode" works too).
#
# MATCH (model paths as the game names them: objects/manmade/vehicles/wagons/wagon_c_hay.cgf)
#   objects/natural/stones/                    a folder: every model under it
#   objects/manmade/structures/logistical/fences/stone_fence
#                                              the start of a file name: stone_fence_a.cgf, stone_fence_b_lod1.cgf ...
#   objects/manmade/vehicles/wagons/wagon_c_hay.cgf
#                                              one model (a match that ends in .cgf, .cga, .chr, .cdf or .skin)
#   *hlod*                                     every model whose path contains this text
#   @actor  @animated  @nodraw  @merged  @unnamed
#                                              what the game says the object is (see "Object states")
#   @animated objects/mymod/windmill.cgf       a state, for these models only (@actor, @animated and @nodraw)
#   @default                                   everything that no other rule matches (one line, in this file or in
#                                              KCD2_TPVCamera_rules_User.txt)
#
# BEHAVIOUR
#   collide    the camera stops at its collision, as at a wall. It never fades.
#   fade       the camera passes through it, and it dithers out while it hides your character (when it is wider than
#              a post).
#   tree       the camera passes through it. The see-through window opens it in front of your character. It never
#              fades whole.
#   bush       like tree, but it also fades whole when the camera ends up inside it.
#   grass      like tree (the log says grass).
#   frame      the camera passes through it. It never fades. Only the part right at the lens thins away.
#   pass       the camera passes through it, and nothing else (invisible collision boxes).
#   character  the camera passes through it. It never fades (people, animals and what they carry).
#   exclude    TPVCamera leaves it out. It never fades, never opens in the window and never thins at the lens, and its
#              collision stops the camera like any collision TPVCamera knows nothing about. For a model that must
#              always stop the camera, write collide: an excluded model also gives up its claim on the level collision
#              around it, and a model next to it can take that collision over.
#
# OPTIONS (after the behaviour)
#   mesh             where the rule ends in collide, the camera also stops at the drawn model where it has no
#                    collision (cloth canopies, thatch under the roof collision), and passes through collision where
#                    the model draws nothing (the invisible canvas walls of a tent). For buildings, not for rocks or
#                    ground.
#   if-wall=<b>      measured in game: a model taller than 1.8 m whose upper part no one sees through gets <b>. A low
#                    or open one keeps the behaviour. Until it is measured, or when it cannot be measured, <b>.
#   if-bare=<b>      a model that draws no cloth (no two-sided part) gets <b>. A model that draws cloth keeps the
#                    behaviour. A rule takes if-wall or if-bare, not both. <b> is one of collide, fade, tree, bush,
#                    grass, frame and pass.
#   only=<kinds>     the rule counts only for objects placed as these kinds (a comma list):
#                      brush       level scenery            movable     movable level scenery
#                      entity      game objects (drivable carts, doors, chests ...)
#                      vegetation  placed with the vegetation tool
#   twins=character  prepare the close-up fade's shaders for the materials of these models at the next pause or
#                    loading screen. It applies whatever rule decides the camera for the model.
#
# WHAT THE RULES CANNOT CHANGE
#   The camera meets only collision fixed in the world, and the ground. Objects the game moves by physics (crates and
#   barrels, and probably the drivable carts) never stop it, whatever their rule: collide only
#   keeps them from fading (with mesh the camera may stop at their drawn model), and exclude
#   lets the camera through them. A person or an animal has no body the camera meets, only a few small static
#   colliders (about 0.3 to 0.7 m across) that the camera's rays meet now and then. With @actor = collide the camera
#   would stop at a person only where a ray happens to meet one of those. The @actor line of this file says
#   character, which lets the camera pass them.
#   Collision built into the level is matched to its model by its box, and tree, bush, grass and character models
#   never take that match. A model placed as level scenery (brush) with one of those words still stops the camera at
#   its built-in collision. Plants placed with the vegetation tool and game objects (entity) carry their own
#   collision, and the camera passes them as their rule says.
#
# WHICH RULE WINS: the first of these that matches the object.
#   1. a state with a model path   (@animated objects/...)      in one file, the longest path wins
#   2. @actor or @animated         (people, animals, what they carry, objects that play an animation)
#   3. a text match                (*...*)                      in one file, the longest text wins
#   4. any other state             (@nodraw, @merged, @unnamed)
#   5. a model path                                             in one file, the longest path wins
#   6. @default
#   A text match beats every model path, even the path of one whole model, so keep text matches rare. A level always
#   comes first, whatever the file: a model path in your own file never beats this file's @nodraw, @actor or @animated
#   line, its @nodraw tents line or its text matches (*hlod*, *designer_*). At the same level, a later file wins over
#   an earlier one, whatever the length of its match: a folder line in your own file changes every model under that
#   folder, the models this file names by their whole path included. In one file, the longest match wins, then a rule
#   with only= for fewer kinds, then the later line. KCD2_TPVCamera.log names both lines when a line repeats a match
#   or a later file overrides one, names each model path a text match takes over, and says how many longer lines of
#   earlier files a line of a later file overrides.
#
# MIXED GAME FOLDERS: ONE LINE PER MODEL
#   Most of the game's folders hold one kind of object, and one folder line decides them all. A few folders hold big
#   structures and small props together (pillory platforms with pillory posts, broad haystacks with drying poles,
#   covered wagons with loose wheels, stone walls with wooden fences). Such a folder has no plain folder line: every
#   model in it has a line with its whole path, and no model line overrides a line of its own folder. A model such a
#   folder holds but this file does not list (one a mod or a game update adds) falls back to the line of its nearest
#   parent folder that has one:
#     objects/manmade/structures/municipal/law_and_order/            objects/manmade/ = collide mesh
#     objects/manmade/structures/industrial/hay/                     objects/manmade/ = collide mesh
#     objects/manmade/vehicles/wagons/                               objects/manmade/vehicles/ = fade
#     objects/manmade/structures/scafolding/                         objects/manmade/ = collide mesh
#     objects/manmade/structures/logistical/fences/                  objects/manmade/ = collide mesh
#     objects/manmade/structures/logistical/gate/                    objects/manmade/ = collide mesh
#     objects/manmade/structures/living/tents/                       objects/manmade/ = collide mesh
#     objects/manmade/structures/logistical/grandstands/             objects/manmade/ = collide mesh
#     objects/manmade/task_specific_props/metal_industry/mining/     objects/manmade/task_specific_props/ = fade
#     objects/manmade/task_specific_props/wood_industry/loggingcamp/ objects/manmade/task_specific_props/ = fade
#     objects/manmade/task_specific_props/household/firewood/        objects/manmade/task_specific_props/ = fade
#     objects/manmade/task_specific_props/construction/              objects/manmade/task_specific_props/ = fade
#     objects/manmade/task_specific_props/construction/cranes/       objects/manmade/task_specific_props/ = fade
#     objects/manmade/task_specific_props/household/washing/         objects/manmade/task_specific_props/ = fade
#     objects/manmade/quest_specific_props/                          objects/manmade/ = collide mesh
#     objects/natural/animal/                                        @default = collide
#   Under objects/manmade/ = collide mesh a model's line counts for every kind of placement, so a game object
#   (entity) copy of a model listed as collide collides too. To keep such copies fading, add the same path with
#   = fade only=entity,movable,vegetation to KCD2_TPVCamera_rules_User.txt. The lines with only=brush (wagons, wood
#   piles, cranes, the washing pier) leave entity copies to the fade line of a parent folder.
#   To give an unlisted model its own behaviour, write its whole path in KCD2_TPVCamera_rules_User.txt or in the
#   mod's patch:   objects/manmade/structures/industrial/hay/my_hay_pile.cgf = fade
#   To change a whole mixed folder, write its folder line there: a later file wins at the same level, whatever the
#   length of its match, so the line takes every model of the folder and its subfolders, the models listed here
#   included. KCD2_TPVCamera.log then says how many lines of this file it overrides (26 for hay/):
#     objects/manmade/structures/industrial/hay/ = fade
#   The comment on a model line says what the model is and gives its size in metres: x by y by height.
#
# HOW TO FIND THE PATH OF A MODEL
#   Set LogLevel = TRACE in KCD2_TPVCamera.ini, start the game and go up to the object. KCD2_TPVCamera.log then has a
#   line for each model near the camera, once per model:
#     Rules: <model path> [<kind>] -> <behaviour> (<file>:<line> '<match>')
#   It gives the full path, how the object is placed and which rule decided it. TRACE writes a lot, so set LogLevel
#   back to INFO afterwards. What to write in KCD2_TPVCamera_rules_User.txt depends on the match that decided:
#     a model path or @default       the model's path                  objects/x/y.cgf = fade
#     a state ('@animated', '@nodraw objects/...'; the kind shows it too, as in [entity, animated])
#                                    the same state, then the path     @animated objects/x/y.cgf = fade
#                                    A model path alone never beats a state.
#     a text match ('*hlod*')        a text match, such as the whole path between stars:
#                                                                      *objects/x/y.cgf* = collide
#                                    A model path never beats a text match.
#   @merged and @unnamed objects have no model path: only those two state lines change them.
#
# SHADER TWINS: a model or material whose rule can end in tree, bush or grass gets the see-through window's shaders
#   prepared at the next pause or loading screen. One whose rule can end in frame gets the lens-cut shader. Only a rule
#   with a model path or a text match prepares shaders. A plant you add by its path needs nothing else, as long as its
#   material uses the game's Vegetation or Illum shader. A plant the window cannot open still gets the camera
#   behaviour of its rule.
#
# YOUR OWN FILE AND PATCHES: rules in other files beside KCD2_TPVCamera.asi, in the same syntax, add to these.
#   KCD2_TPVCamera_rules_<Name>.txt   patches, from other mods or from you. TPVCamera reads them after this file, in
#                                     name order (upper and lower case alike). At the same level, a later name wins
#                                     over this file and the earlier names, whatever the length of their match.
#   KCD2_TPVCamera_rules_User.txt     your own file. TPVCamera reads it last, and at the same level it wins over every
#                                     other file, whatever the length of their match and whatever their only= says.
#   A mod that ships a patch: copy its KCD2_TPVCamera_rules_<Name>.txt beside KCD2_TPVCamera.asi yourself, and delete
#   it when you remove that mod. Mod managers neither put it there nor remove it. A @default line belongs in this file
#   or in KCD2_TPVCamera_rules_User.txt (TPVCamera ignores one in a patch). Every TPVCamera file here starts with
#   KCD2_TPVCamera, so deleting KCD2_TPVCamera* removes them all.
#
# EDITING WHILE THE GAME RUNS: save KCD2_TPVCamera_rules.txt, a patch or your own file, and TPVCamera reads them all
#   again within about a second. The edit applies only when no line in a file you changed has an error. Until then
#   KCD2_TPVCamera.log lists the lines with an error and the rules in force stay. A line with an error in a file you
#   did not change (a mod's patch written for a later TPVCamera, say) is skipped, as at start. The log names the
#   rules that changed. A model that an edit makes tree, bush, grass or frame opens in the window, or thins at the lens,
#   after the next pause (the Esc menu) or loading screen, where its shaders are prepared.
#
# THE INI STILL APPLIES: UseOccluderFade = false makes the camera stop at everything that has collision.
#   UseTreeSeeThrough = false shuts the window. Menus and conversations shut the window and the lens cut.
#
# EXAMPLES (lines for KCD2_TPVCamera_rules_User.txt)
#   objects/manmade/vehicles/wagons/wagon_c_hay.cgf = fade                    the covered hay wagon fades
#   objects/manmade/structures/industrial/hay/ = collide mesh                 all hay stops the camera
#   objects/manmade/structures/logistical/fences/fence_4rod_01.cgf = collide mesh     one fence always stops the camera
#   objects/manmade/common_furniture/beds/bed_warmer_fancy.cgf = exclude              one model left out of the mod
#   objects/mymod/trees/ = tree                                               a mod's trees open in the window
# =====================================================================================================================

# ---- Object states -------------------------------------------------------------------------------------------------
# These come from the game, not from the model's folder. An object is in at most one state.

@merged             = grass                            # grass and small plants the engine merges into ~16 m patches
@unnamed            = tree                             # vegetation whose model path cannot be read
@nodraw             = pass                             # scenery drawn with one no-draw or collision-proxy material
@actor              = character                        # people, animals, and anything attached to them (a carried bow)
@animated           = collide                          # game objects that play an animation (doors, mills) collide
*hlod*              = exclude     only=brush,movable   # HLOD stand-ins: coarse far copies of whole town blocks
*designer_*         = pass        only=brush,movable   # the level designers' invisible collision boxes


# ---- Everything else -----------------------------------------------------------------------------------------------
# Buildings and everything not listed below stop the camera. Man-made models use their drawn mesh too.

@default                = collide                      # terrain meshes, rocks, cliffs, banks, level scenery, unknown
objects/manmade/        = collide     mesh             # houses, walls, churches, castles, sheds, bridges, stairs ...
objects/intermediates/  = collide     mesh             # building pieces (walls, roofs, beams, roof finials)


# ---- Trees ---------------------------------------------------------------------------------------------------------
# The lumberjack's standing beech and its stumps are listed with the logging camp, under the work props.

objects/natural/vegetation/trees/           = tree     # trees, stumps, roots, fallen trunks


# ---- Bushes and plants ---------------------------------------------------------------------------------------------
# The logging camp's branch debris (the same meshes as vegetation/branches/) is listed with the logging camp.

objects/natural/vegetation/                 = bush     # bushes, branches, ferns, vines, plants, grass meshes
objects/natural/wall_deco/                  = bush     # climbing plants on walls (the folder holds only them)


# ---- Grass ---------------------------------------------------------------------------------------------------------

objects/natural/special/                    = grass    # grass meshes placed as one object
objects/natural/special/m01_fake_terrain/   = collide  # a far terrain mesh filed beside them: not grass


# ---- Furniture, decorations, lamps, tools, ladders -----------------------------------------------------------------

objects/manmade/common_furniture/           = fade
objects/manmade/common_decorations/         = fade     # (ground_deco/ is ground, below)
objects/manmade/common_illumination/        = fade
objects/manmade/common_tools/               = fade
objects/manmade/common_fixtures/ladders/    = fade     # other fixtures (doors, windows, chimneys, stairs) collide


# ---- Ground: stones, ground decoration, water decoration -----------------------------------------------------------
# Ground meshes collide like the terrain and the rocks (@default): their own collision, where they have any, stops the
# camera, and they never fade. No mesh: a decal, a mound or a flat stone has no wall that its drawn mesh should add.
# The camp ground around tents, the animals' mounds and the dirt mounds of construction/ are ground too (listed with
# the tents, the animal traces and the work props below).

objects/natural/stones/                             = collide   # stones and boulders up to 1.5 m
objects/natural/terrain_deco/                       = collide   # field banks, road stones, leaf piles, puddles, roots
objects/natural/water_deco/                         = collide   # duckweed and leaves on water, swamp soil patches
objects/manmade/common_decorations/ground_deco/     = collide   # floor stones, rubble mounds, plank curbs, drains

# ---- Tents and canopies --------------------------------------------------------------------------------------------
# A tent, canopy or curtain that draws cloth stops the camera, and its cloth check makes it a frame the camera passes
# while its cloth is hidden. Bare frames, poles, pegs and ropes are frames. Hanging decorations without collision
# (pennants, draped cloth) fade. The camp ground, roads, straw and turf around the tents are ground.
# The @nodraw line gives a tent model drawn with one no-draw material the cloth check of the tent lines below. It also
# prepares the lens-cut shaders of every tent material and LOD file by name: keep its match if you override it.
